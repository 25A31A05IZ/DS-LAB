to find the number of repeatations using linear search
#include <iostream>
using namespace std;

int main() {
    int arr[] = {4, 2, 5, 2, 7, 2, 8, 2, 9, 1};
    int size = sizeof(arr) / sizeof(arr[0]);
    int key, count = 0;
    cout << "Array elements: ";
    for (int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
    cout << "Enter the key to search: ";
    cin >> key;
    for (int i = 0; i < size; i++) {
        if (arr[i] == key) {
            count++;
        }
    }
    if (count > 0) {
        cout << "The key " << key << " is repeated " << count << " time(s)." << endl;
    } else {
        cout << "The key " << key << " is not found in the array." << endl;
    }

    return 0;
}
